  <script>
  // --- State ---
  let allRecords = [];
  let pendingDelete = null;

  // --- Element SDK ---
  const defaultConfig = {
    form_title: 'Data Entry',
    subtitle_text: 'Add and manage your records below.',
    background_color: '#0f1117',
    surface_color: '#181a24',
    text_color: '#f0f0f0',
    primary_action_color: '#6366f1',
    secondary_action_color: '#6b7280',
    font_family: 'DM Sans',
    font_size: 16
  };

  function applyConfig(c) {
    const bg = c.background_color || defaultConfig.background_color;
    const sf = c.surface_color || defaultConfig.surface_color;
    const tx = c.text_color || defaultConfig.text_color;
    const pa = c.primary_action_color || defaultConfig.primary_action_color;
    const sa = c.secondary_action_color || defaultConfig.secondary_action_color;
    const ff = (c.font_family || defaultConfig.font_family) + ', sans-serif';
    const fs = c.font_size || defaultConfig.font_size;

    document.getElementById('app').style.backgroundColor = bg;
    document.querySelectorAll('input,select,textarea').forEach(el => { el.style.backgroundColor = bg; el.style.color = tx; });
    document.querySelectorAll('form, #empty-state, #delete-overlay > div').forEach(el => el.style.backgroundColor = sf);
    document.getElementById('el-title').textContent = c.form_title || defaultConfig.form_title;
    document.getElementById('el-title').style.color = tx;
    document.getElementById('el-subtitle').textContent = c.subtitle_text || defaultConfig.subtitle_text;
    document.getElementById('el-subtitle').style.color = sa;
    document.getElementById('submit-btn').style.backgroundColor = pa;
    document.getElementById('count-badge').style.color = sa;
    document.getElementById('empty-state').style.color = sa;

    document.body.style.fontFamily = ff;
    document.getElementById('el-title').style.fontSize = (fs * 1.9) + 'px';
    document.getElementById('el-subtitle').style.fontSize = fs + 'px';

    // Update record cards
    document.querySelectorAll('.rec-card').forEach(card => {
      card.style.backgroundColor = sf;
      card.querySelector('.rec-name').style.color = tx;
      card.querySelector('.rec-email').style.color = sa;
      card.querySelector('.rec-cat').style.backgroundColor = pa + '20';
      card.querySelector('.rec-cat').style.color = pa;
    });
  }

  window.elementSdk.init({
    defaultConfig,
    onConfigChange: async (config) => applyConfig(config),
    mapToCapabilities: (config) => ({
      recolorables: [
        { get: () => config.background_color || defaultConfig.background_color, set: v => { config.background_color = v; window.elementSdk.setConfig({ background_color: v }); } },
        { get: () => config.surface_color || defaultConfig.surface_color, set: v => { config.surface_color = v; window.elementSdk.setConfig({ surface_color: v }); } },
        { get: () => config.text_color || defaultConfig.text_color, set: v => { config.text_color = v; window.elementSdk.setConfig({ text_color: v }); } },
        { get: () => config.primary_action_color || defaultConfig.primary_action_color, set: v => { config.primary_action_color = v; window.elementSdk.setConfig({ primary_action_color: v }); } },
        { get: () => config.secondary_action_color || defaultConfig.secondary_action_color, set: v => { config.secondary_action_color = v; window.elementSdk.setConfig({ secondary_action_color: v }); } },
      ],
      borderables: [],
      fontEditable: { get: () => config.font_family || defaultConfig.font_family, set: v => { config.font_family = v; window.elementSdk.setConfig({ font_family: v }); } },
      fontSizeable: { get: () => config.font_size || defaultConfig.font_size, set: v => { config.font_size = v; window.elementSdk.setConfig({ font_size: v }); } }
    }),
    mapToEditPanelValues: (config) => new Map([
      ['form_title', config.form_title || defaultConfig.form_title],
      ['subtitle_text', config.subtitle_text || defaultConfig.subtitle_text]
    ])
  });

  // --- Data SDK ---
  const dataHandler = {
    onDataChanged(data) {
      allRecords = data;
      renderRecords();
    }
  };

  window.dataSdk.init(dataHandler);

  // --- Render ---
  function renderRecords() {
    const list = document.getElementById('records-list');
    const empty = document.getElementById('empty-state');
    document.getElementById('count-badge').textContent = `(${allRecords.length})`;
    document.getElementById('limit-warn').classList.toggle('hidden', allRecords.length < 999);

    if (!allRecords.length) { empty.classList.remove('hidden'); list.innerHTML = ''; return; }
    empty.classList.add('hidden');

    const existingMap = new Map([...list.children].map(el => [el.dataset.id, el]));
    const ids = new Set(allRecords.map(r => r.__backendId));

    // remove deleted
    existingMap.forEach((el, id) => { if (!ids.has(id)) el.remove(); });

    const cfg = window.elementSdk.config || defaultConfig;
    const sf = cfg.surface_color || defaultConfig.surface_color;
    const tx = cfg.text_color || defaultConfig.text_color;
    const sa = cfg.secondary_action_color || defaultConfig.secondary_action_color;
    const pa = cfg.primary_action_color || defaultConfig.primary_action_color;

    allRecords.forEach(r => {
      let card = existingMap.get(r.__backendId);
      if (!card) {
        card = document.createElement('div');
        card.className = 'rec-card rounded-xl p-4 border flex items-start justify-between fade-in';
        card.style.borderColor = '#2a2d3a';
        card.dataset.id = r.__backendId;
        card.innerHTML = `
          <div class="min-w-0 flex-1">
            <p class="rec-name font-semibold text-sm truncate"></p>
            <p class="rec-email text-xs mt-0.5 truncate"></p>
            <span class="rec-cat inline-block text-xs font-medium px-2 py-0.5 rounded-full mt-2"></span>
            <p class="rec-notes text-xs mt-2 line-clamp-2" style="color:${sa};"></p>
          </div>
          <button class="rec-del ml-3 mt-1 p-1.5 rounded-lg hover:opacity-80 transition-opacity" style="color:${sa};">
            <i data-lucide="trash-2" style="width:15px;height:15px;"></i>
          </button>`;
        card.querySelector('.rec-del').addEventListener('click', () => showDeleteConfirm(r));
        list.appendChild(card);
        lucide.createIcons();
      }
      card.style.backgroundColor = sf;
      card.querySelector('.rec-name').textContent = r.name;
      card.querySelector('.rec-name').style.color = tx;
      card.querySelector('.rec-email').textContent = r.email;
      card.querySelector('.rec-email').style.color = sa;
      card.querySelector('.rec-cat').textContent = r.category;
      card.querySelector('.rec-cat').style.backgroundColor = pa + '20';
      card.querySelector('.rec-cat').style.color = pa;
      const notesEl = card.querySelector('.rec-notes');
      if (r.notes) { notesEl.textContent = r.notes; notesEl.classList.remove('hidden'); }
      else { notesEl.classList.add('hidden'); }
    });
  }

  // --- Toast ---
  function showToast(msg) {
    const t = document.getElementById('toast');
    document.getElementById('toast-msg').textContent = msg;
    t.classList.remove('hidden');
    t.querySelector('div').classList.remove('toast-slide');
    void t.querySelector('div').offsetWidth;
    t.querySelector('div').classList.add('toast-slide');
    setTimeout(() => t.classList.add('hidden'), 2500);
  }

  // --- Form Submit ---
  document.getElementById('entry-form').addEventListener('submit', async (e) => {
    e.preventDefault();
    if (allRecords.length >= 999) return;
    const btn = document.getElementById('submit-btn');
    const origText = document.getElementById('submit-text').textContent;
    btn.disabled = true; btn.style.opacity = '0.6';
    document.getElementById('submit-text').textContent = 'Saving…';

    const result = await window.dataSdk.create({
      name: document.getElementById('f-name').value.trim(),
      email: document.getElementById('f-email').value.trim(),
      category: document.getElementById('f-cat').value,
      notes: document.getElementById('f-notes').value.trim(),
      created_at: new Date().toISOString()
    });

    btn.disabled = false; btn.style.opacity = '1';
    document.getElementById('submit-text').textContent = origText;

    if (result.isOk) {
      e.target.reset();
      showToast('Record added!');
    } else {
      showToast('Failed to save. Try again.');
    }
  });

  // --- Delete ---
  function showDeleteConfirm(record) {
    pendingDelete = record;
    document.getElementById('delete-overlay').classList.remove('hidden');
  }
  document.getElementById('del-cancel').addEventListener('click', () => {
    document.getElementById('delete-overlay').classList.add('hidden');
    pendingDelete = null;
  });
  document.getElementById('del-confirm').addEventListener('click', async () => {
    if (!pendingDelete) return;
    const btn = document.getElementById('del-confirm');
    btn.disabled = true; btn.textContent = 'Deleting…';
    const result = await window.dataSdk.delete(pendingDelete);
    btn.disabled = false; btn.textContent = 'Delete';
    document.getElementById('delete-overlay').classList.add('hidden');
    if (result.isOk) showToast('Record deleted.');
    else showToast('Failed to delete. Try again.');
    pendingDelete = null;
  });

  lucide.createIcons();
</script>
